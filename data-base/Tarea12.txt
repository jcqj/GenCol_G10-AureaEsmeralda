
CREATE DATABASE aurea_esmeralda
    WITH
    OWNER = postgres
    ENCODING = 'UTF8'
    LOCALE_PROVIDER = 'libc'
    CONNECTION LIMIT = -1
    IS_TEMPLATE = False;
	

-- 1 TABLA USUARIOS
CREATE TABLE public.usuarios
(
	id_us SERIAL,
	nombre_us VARCHAR(150) NOT NULL,
	telefono_us VARCHAR(15) NOT NULL,
	correo_us VARCHAR(150) NOT NULL,
	contrasena_us VARCHAR(150) NOT NULL,
	estado_activo_us BOOLEAN DEFAULT TRUE,
	rol_us VARCHAR(20),
	
	CONSTRAINT pk_id_us PRIMARY KEY (id_us),
	CONSTRAINT uq_correo_us UNIQUE (correo_us),
	CONSTRAINT ck_rol_us CHECK (rol_us IN ('cliente' , 'admin'))
);

-- 2 TABLA CATEGORIAS
CREATE TABLE public.categorias
(
    id_cat SERIAL,
    nombre_cat VARCHAR(100) NOT NULL,
    descripcion_cat TEXT,

    CONSTRAINT pk_id_cat PRIMARY KEY (id_cat),
    CONSTRAINT uq_nombre_cat UNIQUE (nombre_cat)
);

-- 3 TABLA PRODUCTOS
CREATE TABLE public.productos
(
    id_pd SERIAL,
    cat_id INTEGER,
    nombre_pd VARCHAR(150) NOT NULL,
    descripcion_pd TEXT,
    precio_original_pd DECIMAL(12,0) NOT NULL,
    descuento_pd DECIMAL(5,0) DEFAULT 0,
    stock_pd INTEGER DEFAULT 0,
    disponible_pd BOOLEAN DEFAULT TRUE,
    best_seller_pd BOOLEAN DEFAULT FALSE,
    imagen_principal_pd VARCHAR(255),
    imagen_secundaria_pd VARCHAR(255),
    creado_en_pd TIMESTAMP DEFAULT NOW(),

    CONSTRAINT pk_id_pd PRIMARY KEY (id_pd),
    CONSTRAINT fk_cat_id FOREIGN KEY (cat_id)
        REFERENCES categorias(id_cat) ON DELETE SET NULL,
    CONSTRAINT ck_precio_original CHECK (precio_original_pd >= 0),
    CONSTRAINT ck_descuento CHECK (descuento_pd >= 0 AND descuento_pd <= 100),
    CONSTRAINT ck_stock CHECK (stock_pd >= 0)
);

CREATE INDEX idx_productos_cat ON productos(cat_id);
CREATE INDEX idx_productos_disponible ON productos(disponible_pd);
CREATE INDEX idx_productos_best_seller ON productos(best_seller_pd);

-- 4 CERTIFICADOS
CREATE TABLE public.certificados
(
    id_cert SERIAL,
    pd_id INTEGER NOT NULL,
    nombre_gemologo_cert VARCHAR(150),
    creado_en_cert TIMESTAMP DEFAULT NOW(),
    detalles_cert TEXT,
    codigo_cert VARCHAR(50) NOT NULL,
    archivo_url_cert VARCHAR(255),
    imagen_joya_cert VARCHAR(255),

    CONSTRAINT pk_id_cert PRIMARY KEY (id_cert),
    CONSTRAINT fk_cert_pd FOREIGN KEY (pd_id)
        REFERENCES productos(id_pd) ON DELETE CASCADE,
    CONSTRAINT uq_codigo_cert UNIQUE (codigo_cert)
);

CREATE INDEX idx_certificados_producto ON certificados(pd_id);

-- 5 FAVORITOS
CREATE TABLE public.favoritos
(
    id_fav SERIAL,
    us_id INTEGER NOT NULL,
    pd_id INTEGER NOT NULL,
    creado_en_fav TIMESTAMP DEFAULT NOW(),

    CONSTRAINT pk_id_fav PRIMARY KEY (id_fav),
    CONSTRAINT fk_fav_us FOREIGN KEY (us_id)        
        REFERENCES usuarios(id_us) ON DELETE CASCADE,
    CONSTRAINT fk_fav_pd FOREIGN KEY (pd_id)      
        REFERENCES productos(id_pd) ON DELETE CASCADE,
    CONSTRAINT uq_fav_us_pd UNIQUE (us_id, pd_id)
);

-- 6 CARRITOS
CREATE TABLE public.carritos
(
    id_car SERIAL,
    us_id INTEGER NOT NULL,
    creado_en_car TIMESTAMP DEFAULT NOW(),
    actualizado_en_car TIMESTAMP,

    CONSTRAINT pk_id_car PRIMARY KEY (id_car),
    CONSTRAINT fk_car_us FOREIGN KEY (us_id)      
        REFERENCES usuarios(id_us) ON DELETE CASCADE,
    CONSTRAINT uq_us_id UNIQUE (us_id)
);

-- 7 CARRITO ITEMS
CREATE TABLE public.carrito_items
(
    id_it SERIAL,
    car_id INTEGER NOT NULL,
    pd_id INTEGER NOT NULL,
    precio_momentaneo_it DECIMAL(12,0),
    cantidad_it INTEGER DEFAULT 1,
    creado_en_it TIMESTAMP DEFAULT NOW(),

    CONSTRAINT pk_id_it PRIMARY KEY (id_it),
    CONSTRAINT fk_car_id FOREIGN KEY (car_id)
        REFERENCES carritos(id_car) ON DELETE CASCADE,
    CONSTRAINT fk_items_producto FOREIGN KEY (pd_id)
        REFERENCES productos(id_pd),              
    CONSTRAINT ck_cantidad CHECK (cantidad_it > 0),
    CONSTRAINT uq_items_carrito_producto UNIQUE (car_id, pd_id)
);

CREATE INDEX idx_items_carrito ON carrito_items(car_id);
CREATE INDEX idx_items_producto ON carrito_items(pd_id);
